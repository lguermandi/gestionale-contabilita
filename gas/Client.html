const chartSvg = document.getElementById('storico-chart'); const chartLegend = document.getElementById('storico-legend'); let sessionToken = sessionStorage.getItem(SESSION_KEY) || ''; function callLogin(username, password) { return new Promise((resolve, reject) => { google.script.run.withSuccessHandler(resolve).withFailureHandler(reject).login(username, password); [...revenues, ...costs].forEach(row => { MONTHS.forEach(month => profit[month] -= row[month]); profit.totale_anno -= row.totale_anno; }); const addSection = (label, rows, color, background) => { const heading = document.createElement('tr'); const title = makeCell(label, { backgroundColor: background, color: color, fontWeight: 'bold', textTransform: 'uppercase' }); title.colSpan = 14; heading.appendChild(title); body.appendChild(heading); rows.forEach(row => { const tr = document.createElement('tr'); tr.appendChild(makeCell(row.conto || '', { textAlign: 'left', borderLeft: `3px solid ${color}` })); MONTHS.forEach(month => { const value = row[month]; tr.appendChild(makeCell(value === 0 ? '-' : formatNumber(Math.abs(value)), { color: '#a0a0a0' })); }); tr.appendChild(makeCell(formatNumber(Math.abs(row.totale_anno)), { fontWeight: 'bold', color: '#fff' })); body.appendChild(tr); }); }; addSection('Entrate / Ricavi', revenues, '#4caf50', 'rgba(76, 175, 80, .15)'); addSection('Uscite / Costi', costs, '#f44336', 'rgba(244, 67, 54, .15)'); const profitRow = document.createElement('tr'); profitRow.style.borderTop = '2px solid #666'; profitRow.appendChild(makeCell('Utile Mensile', { textAlign: 'left', fontWeight: 'bold', color: '#fff', textTransform: 'uppercase' })); MONTHS.forEach(month => { const value = profit[month]; profitRow.appendChild(makeCell(value === 0 ? '-' : formatNumber(value), { fontWeight: 'bold', color: value > 0 ? '#4caf50' : value < 0 ? '#f44336' : '#a0a0a0' })); }); profitRow.appendChild(makeCell(formatNumber(profit.totale_anno), { fontWeight: 'bold', color: profit.totale_anno > 0 ? '#4caf50' : profit.totale_anno < 0 ? '#f44336' : '#a0a0a0' })); body.appendChild(profitRow); } function renderGestioni(gestioni) { const body = document.getElementById('corpo-gestioni'); body.replaceChildren(); gestioni.forEach(item => { const row = document.createElement('tr'); row.appendChild(makeCell(item.gestione || '', { textAlign: 'left', textTransform: 'capitalize', fontWeight: 'bold' })); row.appendChild(makeCell(formatNumber(item.totale_ricavi), { color: '#a0a0a0' })); row.appendChild(makeCell(formatNumber(item.totale_costi), { color: '#a0a0a0' })); row.appendChild(makeCell(formatNumber(item.utile), { fontWeight: 'bold', color: item.utile >= 0 ? '#4caf50' : '#f44336'; item.append(description, amount); list.appendChild(item); }); } function svgElement(name, attributes) { const element = document.createElementNS('http://www.w3.org/2000/svg', name); Object.keys(attributes).forEach(key => element.setAttribute(key, attributes[key])); return element; } function renderChart(timeline) { chartSvg.replaceChildren(); chartLegend.replaceChildren(); const width = 1000; const height = 400; const padding = { left: 86, right: 20, top: 18, bottom: 38 }; chartSvg.setAttribute('viewBox', `0 0 ${width} ${height}`); if (!timeline.length) { const message = svgElement('text', { x: width / 2, y: height / 2, fill: '#aaa', 'text-anchor': 'middle' }); message.textContent = 'Nessun movimento registrato'; chartSvg.appendChild(message); return; } const accounts = [...new Set(timeline.flatMap(day => Object.keys(day).filter(key => key !== 'data')))]; const values = timeline.flatMap(day => accounts.map(account => Number(day[account]) || 0)); const minimum = Math.min(0, ...values); const maximum = Math.max(0, ...values); const range = maximum - minimum || 1; const plotWidth = width - padding.left - padding.right; const plotHeight = height - padding.top - padding.bottom; const x = index => padding.left + (timeline.length === 1 ? plotWidth / 2 : index * plotWidth / (timeline.length - 1)); const y = value => padding.top + (maximum - value) * plotHeight / range; for (let tick = 0; tick <= 4; tick += 1) { const value = maximum - range * tick / 4; const position = y(value); chartSvg.appendChild(svgElement('line', { x1: padding.left, y1: position, x2: width - padding.right, y2: position, stroke: '#333', 'stroke-width': 1 })); const label = svgElement('text', { x: padding.left - 10, y: position + 4, fill: '#a0a0a0', 'text-anchor': 'end', 'font-size': 11 }); label.textContent = formatNumber(Math.abs(value)); chartSvg.appendChild(label); } const labelCount = Math.min(7, timeline.length); for (let index = 0; index < labelCount; index += 1) { const pointIndex = labelCount === 1 ? 0 : Math.round(index * (timeline.length - 1) / (labelCount - 1)); const label = svgElement('text', { x: x(pointIndex), y: height - 10, fill: '#a0a0a0', 'text-anchor': 'middle', 'font-size': 11 }); label.textContent = timeline[pointIndex].data; chartSvg.appendChild(label); } accounts.forEach((account, accountIndex) => { const color = `hsl(${accountIndex * 137.5 % 360}, 70%, 60%)`; const points = timeline.map((day, index) => `${x(index)},${y(Number(day[account]) || 0)}`).join(' '); const line = svgElement('polyline', { points: points, fill: 'none', stroke: color, 'stroke-width': 2, 'stroke-linejoin': 'round', 'stroke-linecap': 'round' }); const title = svgElement('title', {}); title.textContent = account; line.appendChild(title); chartSvg.appendChild(line); const legendItem = document.createElement('span'); legendItem.className = 'legend-item'; const swatch = document.createElement('span'); swatch.className = 'legend-swatch'; swatch.style.backgroundColor = color; const name = document.createElement('span'); name.textContent = account; legendItem.append(swatch, name); chartLegend.appendChild(legendItem); }); } async function loadDashboard() { setStatus('Caricamento in corso...', '#ff9800'); try { const data = await callDashboardData(sessionToken, yearSelect.value); renderChart(data.timeline); renderMonthlyReport(data.report); renderGestioni(data.gestioni); renderPositions(data.dettaglio); setStatus('Dati aggiornati', '#4caf50'); } catch (error) { const message = String(error && error.message || error); if (message.includes('SESSION_EXPIRED')) { showLogin('Sessione scaduta.
